cmake_minimum_required(VERSION 3.21)
project(navgen VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(NAVGEN_HARDWARE_FMA "Emit hardware FMA instructions (x86-64 AVX2 required)" ON)
option(NAVGEN_BUILD_TESTS "Build differential module probes and native tests" ON)
if(MSVC)
    add_compile_options(/fp:precise /Oi /utf-8)
    if(NAVGEN_HARDWARE_FMA)
        add_compile_options(/arch:AVX2)
    endif()
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
    add_compile_options(-ffp-contract=off -fno-fast-math)
    if(NAVGEN_HARDWARE_FMA AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
        add_compile_options(-mfma)
    endif()
endif()
find_package(Threads REQUIRED)

file(GLOB QHULL_SOURCES CONFIGURE_DEPENDS third_party/qhull/*_r.c)
add_library(navgen_qhull STATIC ${QHULL_SOURCES})
target_include_directories(navgen_qhull PUBLIC third_party/qhull)
add_library(navgen_zlib STATIC third_party/zlib/adler32.c third_party/zlib/crc32.c
    third_party/zlib/inflate.c third_party/zlib/inffast.c third_party/zlib/inftrees.c
    third_party/zlib/zutil.c third_party/zlib/uncompr.c)
target_include_directories(navgen_zlib PUBLIC third_party/zlib)
add_library(navgen_lzma STATIC third_party/lzma/LzmaDec.c)
target_include_directories(navgen_lzma PUBLIC third_party/lzma)
add_library(navgen_core STATIC src/common.cpp src/navio.cpp src/bsp.cpp src/gma.cpp src/planes.cpp
    src/collision.cpp src/sampling.cpp src/areas.cpp src/hiding.cpp src/merge.cpp
    src/discovery.cpp src/ladders.cpp src/geometry.cpp src/steam.cpp)
target_include_directories(navgen_core PUBLIC src third_party/json/include)
target_link_libraries(navgen_core PUBLIC navgen_qhull navgen_zlib navgen_lzma Threads::Threads)
file(GLOB BUILD_ID_SOURCES CONFIGURE_DEPENDS src/*.hpp src/*.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${BUILD_ID_SOURCES})
set(BUILD_ID_INPUT "navgen-cpp20")
foreach(INPUT IN LISTS BUILD_ID_SOURCES)
    file(SHA256 "${INPUT}" INPUT_HASH)
    string(APPEND BUILD_ID_INPUT "${INPUT_HASH}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" CMAKE_HASH)
string(APPEND BUILD_ID_INPUT "${CMAKE_HASH}")
string(SHA256 BUILD_ID "${BUILD_ID_INPUT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build_id.hpp" "#define NAVGEN_BUILD_ID \"cpp20-${BUILD_ID}\"\n")
add_executable(navgen src/main.cpp src/generate.cpp)
target_include_directories(navgen PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(navgen PRIVATE navgen_core)
if(NAVGEN_BUILD_TESTS)
    enable_testing()
    add_executable(navgen_module_probe tests/module_probe.cpp)
    target_link_libraries(navgen_module_probe PRIVATE navgen_core)
    add_executable(navgen_core_tests tests/core_tests.cpp)
    target_link_libraries(navgen_core_tests PRIVATE navgen_core)
    add_test(NAME native_core COMMAND navgen_core_tests)
    add_executable(navgen_sampling_areas_tests tests/test_sampling_areas.cpp)
    target_link_libraries(navgen_sampling_areas_tests PRIVATE navgen_core)
    add_test(NAME native_sampling_areas COMMAND navgen_sampling_areas_tests)
    add_executable(navgen_optional_environment_probe tests/optional_environment_probe.cpp)
    target_link_libraries(navgen_optional_environment_probe PRIVATE navgen_core)
endif()
